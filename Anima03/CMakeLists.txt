cmake_minimum_required(VERSION 3.21)
project(Anima03 LANGUAGES CXX)

find_package(Threads REQUIRED)
add_executable(anima src/main.cpp src/AnimaBytecode.h src/AnimaGpu.cpp src/AnimaGpu.h)
target_compile_features(anima PRIVATE cxx_std_20)
target_link_libraries(anima PRIVATE Threads::Threads)
target_include_directories(anima PRIVATE src)
set_target_properties(anima PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(anima PRIVATE /W4 /utf-8 /permissive- /fp:precise)
  target_compile_definitions(anima PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(anima PRIVATE d3d12 dxgi d3dcompiler dxguid)
else()
  target_compile_options(anima PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_custom_command(TARGET anima POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
          "${CMAKE_CURRENT_SOURCE_DIR}/Examples"
          "$<TARGET_FILE_DIR:anima>/Anima03Examples")

include(CTest)
if(BUILD_TESTING)
  add_test(NAME anima_self_test COMMAND $<TARGET_FILE:anima> self-test)
  add_test(NAME anima_check_fibonacci COMMAND $<TARGET_FILE:anima> check "${CMAKE_CURRENT_SOURCE_DIR}/Examples/fibonacci.ani")
  add_test(NAME anima_run_fibonacci COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/fibonacci.ani")
  add_test(NAME anima_check_machine_inspector COMMAND $<TARGET_FILE:anima> check "${CMAKE_CURRENT_SOURCE_DIR}/Examples/machine-inspector.ani")
  add_test(NAME anima_run_machine_inspector COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/machine-inspector.ani")
  add_test(NAME anima_run_fixed_step COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/fixed-step-simulation.ani")
  add_test(NAME anima_enums_and_records COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/enums-and-records.ani")
  add_test(NAME anima_records_enum_defaults COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/enums-and-records.ani")
  add_test(NAME anima_local_module_import COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/Modules/module-use.ani")
  add_test(NAME anima_local_module_enum_default COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/Modules/enum-default-test.ani")
  add_test(NAME anima_ui_catalog COMMAND $<TARGET_FILE:anima> ui-catalog Button)
  add_test(NAME anima_map_key_value_iteration COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Examples/map-iteration.ani")
  add_test(NAME anima_bytecode_disassembly COMMAND $<TARGET_FILE:anima> dis "${CMAKE_CURRENT_SOURCE_DIR}/Examples/fibonacci.ani")
  add_test(NAME anima_test_blocks COMMAND $<TARGET_FILE:anima> test "${CMAKE_CURRENT_SOURCE_DIR}/Tests/conformance-smoke.ani")
  add_test(NAME anima_compat_02 COMMAND $<TARGET_FILE:anima> run "${CMAKE_CURRENT_SOURCE_DIR}/Reference/0.2-examples/definitions.ani" --compat 0.2)
endif()
