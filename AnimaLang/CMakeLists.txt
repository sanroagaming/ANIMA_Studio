cmake_minimum_required(VERSION 3.24)
project(AnimaLang LANGUAGES CXX)
add_executable(AnimaLang src/AnimaLang.cpp src/AnimaGpu.cpp)
target_compile_features(AnimaLang PRIVATE cxx_std_20)
if(WIN32)
  target_compile_definitions(AnimaLang PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(AnimaLang PRIVATE d3d12 dxgi d3dcompiler dxguid)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(AnimaLang PRIVATE Threads::Threads)
endif()
